# DyTopo Environment Variables
# Copy this file to .env and fill in your values
#
# ── Embedding Provider ──────────────────────────────────────────────────────
#
# DYTOPO_EMBEDDER controls which embedding backend is used.
# Auto-detection order if unset:
#   1. openai     (if OPENAI_API_KEY is present)
#   2. anthropic  (if VOYAGE_API_KEY or ANTHROPIC_API_KEY is present)
#   3. local      (sentence-transformers, no API key needed)
#
# DYTOPO_EMBEDDER=openai

# ── OpenAI (default if key present) ────────────────────────────────────────
# Model: text-embedding-3-small (~$0.00002 / 1K tokens)
# OPENAI_API_KEY=sk-your-key-here

# ── Voyage AI (used when DYTOPO_EMBEDDER=anthropic) ────────────────────────
# Model: voyage-3-lite  Install: pip install voyageai
# Get key at: https://www.voyageai.com/
# VOYAGE_API_KEY=pa-your-voyage-key-here
#
# ANTHROPIC_API_KEY is accepted as a fallback alias for VOYAGE_API_KEY
# ANTHROPIC_API_KEY=sk-ant-your-key-here

# ── Local / sentence-transformers ───────────────────────────────────────────
# No API key needed. Downloads ~80MB model on first use.
# Default model: all-MiniLM-L6-v2
# Override with any sentence-transformers model name:
# DYTOPO_LOCAL_MODEL=all-MiniLM-L6-v2

# ── Hook Settings ───────────────────────────────────────────────────────────
# Number of agents selected by the PreToolUse hook (default: 3)
# DYTOPO_HOOK_K=3
#
# Graph type used by hook routing (default: knn)
# DYTOPO_HOOK_GRAPH=knn
#
# Path to agents.yaml (default: same directory as dytopo_hook.py)
# DYTOPO_AGENTS_YAML=/path/to/agents.yaml
